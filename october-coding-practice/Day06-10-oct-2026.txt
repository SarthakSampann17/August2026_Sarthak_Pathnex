# Jenkins file — With Git Integration and Testing

pipeline {
    agent any
    stages {
        stage('Clone Repository') {
            steps {
                git 'https://github.com:SarthakSampann17/August2026_Sarthak_Pathnex.git'
            }
        }
        stage('Build') {
            steps {
                echo 'Building project from Git repository'
            }
        }
        stage('Test') {
            steps {
                echo 'Running unit tests'
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
    }
}


# GitLab CI/CD — With Docker Build
stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t sarthak-app .

push:
  stage: push
  script:
    - docker push sarthak-app

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml
